if(NOT DEFINED Python3_EXECUTABLE)
    set(plangd_venv "$ENV{VIRTUAL_ENV}")
    if(NOT plangd_venv)
        set(plangd_venv "${PROJECT_SOURCE_DIR}/.venv")
    endif()
    if(CMAKE_HOST_WIN32)
        set(plangd_venv_python "${plangd_venv}/Scripts/python.exe")
    else()
        set(plangd_venv_python "${plangd_venv}/bin/python")
    endif()
    if(EXISTS "${plangd_venv_python}")
        set(Python3_EXECUTABLE "${plangd_venv_python}")
    endif()
endif()
